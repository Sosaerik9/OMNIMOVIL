// --- SISTEMA DE SEÑALIZACIÓN Y LLAMADAS REALES ENTRE CLIENTES (FIREBASE) --- // Escuchador de llamadas entrantes en tiempo real para el cliente actual function activarEscuchaLlamadasEntrantes(miNumero) { if (!db) return; db.collection("llamadas") .where("destino", "==", miNumero) .where("estado", "==", "marcando") .onSnapshot((snapshot) => { snapshot.docChanges().forEach((change) => { if (change.type === "added") { let datosLlamada = change.doc.data(); let llamadaId = change.doc.id; // Mostrar pantalla/modal de llamada entrante en el teléfono del receptor document.getElementById('caller-id').innerText = datosLlamada.origen; document.getElementById('incoming-call-modal').style.display = 'flex'; // Guardar referencia global de la llamada actual window.llamadaActualId = llamadaId; window.ofertaRemota = JSON.parse(datosLlamada.offerSDP); } }); }); } // Cliente A: Iniciar llamada al Cliente B async function iniciarLlamadaReal() { if (numeroMarcado.length <= 9) { alert("Marca un número de destino válido."); return; } if (saldoActual < 0.10) { alert("Saldo insuficiente. Ingresa a tu portal y recarga."); return; } let numeroDestino = numeroMarcado; document.getElementById('call-target-display').innerText = numeroDestino; document.getElementById('call-status-title').innerText = "🔄 Llamando por la Red IP..."; document.getElementById('call-timer').innerText = "00:00"; document.getElementById('call-active-overlay').style.display = 'flex'; try { if (!localStream) { localStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false }); } peerConnection = new RTCPeerConnection(servers); localStream.getTracks().forEach(track => { peerConnection.addTrack(track, localStream); }); peerConnection.ontrack = (event) => { document.getElementById('remote-audio').srcObject = event.streams[0]; }; // Crear oferta SDP de WebRTC let offer = await peerConnection.createOffer(); await peerConnection.setLocalDescription(offer); // Guardar la oferta en Firebase para que el Cliente B reciba la llamada if (db) { let docRef = await db.collection("llamadas").add({ origen: miNumeroAsignado, destino: numeroDestino, offerSDP: JSON.stringify(offer), estado: "marcando", fecha: firebase.firestore.FieldValue.serverTimestamp() }); window.llamadaActualId = docRef.id; // Escuchar cuando el Cliente B conteste db.collection("llamadas").doc(docRef.id).onSnapshot(doc => { let data = doc.data(); if (data && data.estado === "contestada" && data.answerSDP) { let answer = new RTCSessionDescription(JSON.parse(data.answerSDP)); peerConnection.setRemoteDescription(answer); document.getElementById('call-status-title').innerText = "🔊 En Comunicación (Voz HD)"; descontarSaldoIniciarContador(); } else if (data && data.estado === "colgada") { colgarLlamadaReal(); } }); } } catch (error) { console.error("Error al iniciar llamada:", error); alert("Error al acceder al micrófono o conectar con la red."); colgarLlamadaReal(); } } // Cliente B: Aceptar la llamada entrante y responder async function aceptarLlamadaEntrante() { document.getElementById('incoming-call-modal').style.display = 'none'; document.getElementById('call-status-title').innerText = "🔄 Conectando Audio..."; document.getElementById('call-active-overlay').style.display = 'flex'; try { if (!localStream) { localStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false }); } peerConnection = new RTCPeerConnection(servers); localStream.getTracks().forEach(track => { peerConnection.addTrack(track, localStream); }); peerConnection.ontrack = (event) => { document.getElementById('remote-audio').srcObject = event.streams[0]; }; // Establecer la oferta del Cliente A await peerConnection.setRemoteDescription(new RTCSessionDescription(window.ofertaRemota)); // Crear la respuesta SDP de WebRTC let answer = await peerConnection.createAnswer(); await peerConnection.setLocalDescription(answer); // Actualizar Firebase para notificar al Cliente A que la llamada fue contestada if (db && window.llamadaActualId) { await db.collection("llamadas").doc(window.llamadaActualId).update({ answerSDP: JSON.stringify(answer), estado: "contestada" }); } document.getElementById('call-status-title').innerText = "🔊 En Comunicación (Voz HD)"; descontarSaldoIniciarContador(); } catch (error) { console.error("Error al contestar llamada:", error); colgarLlamadaReal(); } } // Función auxiliar para temporizador y saldo function descontarSaldoIniciarContador() { saldoActual -= 0.10; document.getElementById('saldo-usuario').innerText = `$${saldoActual.toFixed(2)}`; callSeconds = 0; if (callTimerInterval) clearInterval(callTimerInterval); callTimerInterval = setInterval(() => { callSeconds++; let mins = Math.floor(callSeconds / 60).toString().padStart(2, '0'); let secs = (callSeconds % 60).toString().padStart(2, '0'); document.getElementById('call-timer').innerText = `${mins}:${secs}`; }, 1000); }
